\begin{frame}{1. 成長がなぜ応力になるのか}
\small 膨らむこと自体は応力を生まない。\textbf{膨らめないこと}が生む。
\vspace{4pt}
\begin{center}
\begin{tikzpicture}[font=\scriptsize]
  % --- 左：自由に膨らむ ---

    \draw[gray!70, dashed, line width=0.9pt] (0,0) rectangle (1.6,1.6);
    \node[gray!70, font=\tiny, above right] at (0.02,1.62) {元の大きさ};
    \foreach \y in {0.4,0.8,1.2} {
      \draw[-{Stealth[length=1.8mm]}, c3] (0,\y) -- ++(-0.28,0);
      \draw[-{Stealth[length=1.8mm]}, c3] (1.6,\y) -- ++(0.28,0);
    }
    \foreach \x in {0.4,0.8,1.2} {
      \draw[-{Stealth[length=1.8mm]}, c3] (\x,0) -- ++(0,-0.28);
      \draw[-{Stealth[length=1.8mm]}, c3] (\x,1.6) -- ++(0,0.28);
    }
    \node[below=1.3cm of {(0.8,0)}, align=center] {自由に膨らむ\\
      \textcolor{c3}{\textbf{応力ゼロ}}};
  \end{scope}
  % --- 右：下がくっついている ---

\small パン生地が型の中で膨らむのと同じ。バイオフィルムは歯面に固着したまま膨らむので、
\textbf{自由に膨らめない分が応力になる}。
\end{frame}

\begin{frame}{$\alpha$ と $\bm{F}_g=(1+\alpha)\bm{I}$}
\small 覚える式は1つ。
\vspace{6pt}
\begin{center}
\colorbox{c3!10}{\parbox{0.8\linewidth}{\centering
\large $\bm{F}_g=(1+\alpha)\,\bm{I}$\\[4pt]
\small 全方向に $(1+\alpha)$ 倍に\textbf{膨らみたい}}}
\end{center}
\vspace{8pt}
\begin{itemize}\small
  \item $\alpha$ ：膨らみたい量。$\alpha=0.05$ で「5\% 大きくなりたい」
  \item $\alpha$ が大きいほど押し合いが強く、応力も大きい
  \item $\bm{I}$ は単位行列 $\rightarrow$ どの方向にも同じだけ
  \item $\alpha$ は細菌の組成から決まる $\rightarrow$ \textbf{菌が変われば応力が変わる}
\end{itemize}
\vspace{8pt}
\small 「膨らみたい」と「実際に膨らんだ」は別物。実際の変形はまわりが許す分だけで、
その\textbf{差}が応力になる。
\end{frame}

\begin{frame}{$\alpha$ はどこから来るのか（2）組成 $\rightarrow$ 成長}
\small 組成を時間発展させ、その積み上げが $\alpha$ になる。
\vspace{5pt}
\begin{center}
\colorbox{c3!10}{\parbox{0.86\linewidth}{\centering\small
$\varphi_{\text{tot}} = \sum_i \varphi_i\,\psi_i$
\qquad\qquad
$\alpha = k_\alpha \displaystyle\int \varphi_{\text{tot}}\,dt$}}
\end{center}
\vspace{6pt}
\begin{itemize}\small
  \item $\varphi_i$：菌種 $i$ の体積分率、$\psi_i$：その生存率
  \item $\varphi_{\text{tot}}$ は\textbf{生きている菌の総量}。これが成長を駆動する
  \item $\varphi_i$ の時間発展は5種の相互作用モデル。
        パラメータ $\bm{\theta}$（20 成分、中心は相互作用行列 $\bm{A}$）を
        \textbf{TMCMC で較正}
  \item 積分するので、\textbf{同じ組成でも時間が長ければ $\alpha$ は大きい}
\end{itemize}
\vspace{5pt}
\small $\psi_i$ は現状 \textbf{0.999 のプレースホルダ}。
実験側の「生細胞のみ」比は 1 を超える値が出ており、モデルの $\psi_i\in[0,1]$ とは
別物と確認したため、\textbf{代入せずに留保}している。
\end{frame}

\begin{frame}{2. USERMAT とは何をするものか}
\small ANSYS は各点・各ステップで\textbf{何万回も}同じことを聞いてくる。
\vspace{4pt}
\begin{center}
\begin{tikzpicture}[font=\small, node distance=8mm,
  b/.style={draw, rounded corners=3pt, minimum height=11mm, text width=30mm,
            align=center},
  ar/.style={-{Stealth[length=2.4mm]}, thick}]
  \node[b, fill=gray!10] (ans) {ANSYS\\（構造解析）};
  \node[b, fill=c3!18, draw=c3, very thick, right=36mm of ans] (um)
       {\textbf{USERMAT}\\自分で書いた材料則};
  \draw[ar, gray] ([yshift=3mm]ans.east) -- node[above, font=\scriptsize, align=center]
       {この点はこれだけ\\変形している} ([yshift=3mm]um.west);
  \draw[ar, c3] ([yshift=-3mm]um.west) -- node[below, font=\scriptsize, align=center]
       {応力はこれ} ([yshift=-3mm]ans.east);
\end{tikzpicture}
\end{center}
\vspace{2pt}
\begin{itemize}\small
  \item 普通のゴムや金属 $\rightarrow$ ANSYS が答えを持っている
  \item 「\textbf{膨らもうとするゴム}」$\rightarrow$ 持っていない
  \item $\Rightarrow$ その答えを返す部品を自分で書いた。\textbf{これが修論の本体}
\end{itemize}
\vspace{6pt}
\small ANSYS 本体には手を入れていない。「聞かれたら答える」部品の差し替えだけ。
バージョンに依存しないのもそのため。
\end{frame}
