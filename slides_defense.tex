\begin{frame}{Motivation and objective}
\begin{columns}[T]
\begin{column}{0.55\textwidth}
\begin{itemize}
  \item Peri-implantitis affects about one in five implant patients (two meta-analyses)
  \item It is driven by multi-species biofilms; a shift towards pathobionts
        such as \textit{P.~gingivalis} goes with disease
  \item A biofilm is also a mechanical body: where it grows unevenly or
        against a stiff substrate, it carries stresses
\end{itemize}
\vspace{6pt}
\textbf{Objective:} to compute the species composition of a biofilm, its growth and the
resulting stresses consistently within an existing finite element framework.
\end{column}
\begin{column}{0.42\textwidth}
\small
\cbox{p1}{1.6cm}{\textbf{\color{p1}Point model}\\ species composition at a material point\\ \textcolor{gray}{no spatial resolution, no stress}}\\[4pt]
\cbox{p2}{1.6cm}{\textbf{\color{p2}Growth field}\\ spatial growth and growth-induced stress\\ \textcolor{gray}{single biofilm phase, no species}}\\[4pt]
\cbox{c3}{1.6cm}{\textbf{\color{c3}This work}\\ both, evaluated at every Gauss point of an existing ANSYS element}
\end{column}
\end{columns}
\end{frame}

\begin{frame}{Both models from the extended Hamilton principle}
\begin{columns}[T]
\begin{column}{0.5\textwidth}
\textbf{Growth field} (Klempt et al.\ 2024)
\[
\begin{aligned}
&\dot\phi=\beta\nabla^2\phi+k_\alpha\alpha\fterm\\
&\dot c=d\,\nabla^2c-g\,\phi\,\textstyle\sum_i\bar\phi_i\,c,\qquad \dot\alpha=k_\alpha\phi\\
&\mathbf F=\mathbf F_e\mathbf F_g,\quad \mathbf F_g=\alpha\mathbf I,\quad \alpha(0)=1
\end{aligned}
\]
Stress only from $\mathbf F_e$; growth is $\alpha-1$.
\end{column}
\begin{column}{0.48\textwidth}
\textbf{Point model} (Klempt et al.\ 2026)
\[
\dot\Psi+\Delta+C\;\to\;\operatorname*{stat}_{\dot{\bm\phi},\,\dot{\bm\psi},\,\gamma}
\]
$\Psi=-\tfrac12c^*\bar{\bm\phi}^{\mathrm T}\mathbf A\,\bar{\bm\phi}$,
$\bar\phi_i=\phi_i\psi_i$, $C=\gamma(\sum_l\phi_l-1)$\\[2pt]
local form: principle of the minimum of the
dissipation potential (Junker et al.\ 2014, Hackl 1997)\\[2pt]
$\Rightarrow$ $\mathbf A$ symmetric, $2n+2$ equations per point
\end{column}
\end{columns}
\vfill
\begin{columns}[T]
\begin{column}{0.5\textwidth}
\cbox{p2}{1.6cm}{\scriptsize $\phi\in[0,1]$: amount of biofilm; $c$: nutrient concentration;
$\alpha$: growth variable, growth $=\alpha-1$; $\beta$: diffusion coefficient of $\phi$ [mm$^2/T^*$];
$k_\alpha$: growth rate [$1/T^*$]; $d$, $g$: nutrient diffusivity [mm$^2/T^*$], consumption rate [$1/T^*$];
$T^*$: normalised time of Klempt et al.\ 2024}
\end{column}
\begin{column}{0.48\textwidth}
\cbox{p1}{1.6cm}{\scriptsize $\phi_i$, $\psi_i$: volume fraction and viability of species $i$;
$\bar\phi_i$: living fraction; $\Psi$: free energy, $\Delta$: dissipation function,
$C$: constraint function; $c^*$: nutrient level; $\mathbf A$: interaction matrix;
$\gamma$: Lagrange multiplier of $\sum_{l=0}^n\phi_l=1$ ($\phi_0$: void)}
\end{column}
\end{columns}

\end{frame}

\begin{itemize}\setlength{\itemsep}{3pt}
  \item The antibiotic lowers only $\psi_i$, faster for the more sensitive species (B, $b_2=2b_1$); the nutrient

  \item Mixed start: every seed point holds B, so the seed dies; separate start: the A seed next to the nutrient

\end{itemize}

\begin{frame}{Limitations}
\begin{itemize}\setlength{\itemsep}{7pt}
  \item \textbf{Verification, not validation}: no measured stress of a growing
        oral biofilm was available
    \item \textbf{Mechanics one way}: the stress does not act back on growth (as in Klempt et al.\ 2024),
        and the composition does not act on the stiffness
  \item \textbf{Front term \fterm{}}: its speed depends on the nutrient only, so dead biofilm keeps spreading
        after the antibiotic; the stiffness counts dead biomass
  \item \textbf{Assumptions}: start values of newly reached points, $\phi_{\min}$, the conversion
        of $\beta$; each reported with its sensitivity
  \item \textbf{Calibrated parameters in the FE model}: one earlier one-way run (CH) only; Part~3
        uses synthetic data
  \item \textbf{Symmetric $\mathbf A$}: no directed effects; only part of $\mathbf A$
        is determined; different strains in the commensal and dysbiotic data
\end{itemize}
\takeaway{Each limitation is quantified in the thesis where possible; the results are verified, not yet validated.}
\end{frame}

\begin{frame}[label=appB]{Appendix B: growth field and mechanics}
\small
\textbf{Kinematics.} $\mathbf F=\mathbf F_e\mathbf F_g$ with isotropic
$\mathbf F_g=\alpha\mathbf I$. Stress comes only from $\mathbf F_e$. A body
that can grow freely stays stress free; stress appears where growth is
constrained.\\[3pt]
\textbf{Material.} $E(\phi)=(\phi^2+f)\,E$, $E=10$~Pa, $\nu=0.49$
(Klempt 2024, Table~2; their material is incompressible), $f=10^{-3}$;
decks in mm and MPa, stresses quoted in Pa.\\[3pt]
\textbf{Field equations} (Klempt et al.\ 2024, Eqs.~34--36, without the bars):
\[
\dot\phi=\underbrace{\beta\,\nabla^2\phi}_{\text{diffusion}}
+\underbrace{k_\alpha\,\alpha}_{\text{local source}}
-\underbrace{\|\nabla\phi\|\,\frac{r\,c}{k+c}\,\mathbf n_{\nabla\phi}\cdot\mathbf n_{\nabla c}}_{\text{front, inactive in ANSYS}},
\qquad
\dot c=d\,\nabla^2c-g\,\phi\,c,
\qquad
\dot\alpha=k_\alpha\,\phi
\]
\begin{itemize}
  \item $\phi$ and $c$ are fields of the element, updated explicitly with the
        Neighbored Element Method; $\alpha$ is a state variable at the Gauss points
  \item Consumption $g\phi c$ as in the authors' implementation (printed:
        $g\phi$)
\end{itemize}
\end{frame}

\begin{frame}[label=appC]{Appendix C: point model and one coupled substep}
\small
\textbf{Point model} (Klempt et al.\ 2026), per species $i$, no space:
\[
\begin{aligned}
(\eta_{\phi,i}+\eta_i\psi_i^2)\,\dot\phi_i+\eta_i\phi_i\psi_i\,\dot\psi_i
&= c^*\psi_i\textstyle\sum_j A_{ij}\phi_j\psi_j-\gamma-(\text{barrier})\\
\eta_i\,(\phi_i\psi_i\,\dot\phi_i+\phi_i^2\,\dot\psi_i)
&= c^*\phi_i\textstyle\sum_j A_{ij}\phi_j\psi_j-\alpha^* b_i\,\psi_i-(\text{barrier})
\end{aligned}
\]
with $\sum_i\phi_i+\phi_0=1$; the barrier keeps $\phi_i,\psi_i$ in $(0,1)$;
implicit Euler and Newton, $\Delta t=10^{-4}$.\\[4pt]
\textbf{One substep of the coupled ANSYS run}
\begin{enumerate}
  \item The element updates its field $\phi$
  \item At each Gauss point: $\alpha_{n+1}=\alpha_n+k_\alpha\phi\,\Delta t$,
        $\mathbf F_g=\alpha\mathbf I$, stress and tangent
    \item Where $\phi_{3D}\ge\phi_{\min}$: advance the point model by $s\,\Delta t$ ($s=0.1$) with $c^*=c^*_0c$ and
        $\alpha^*=\alpha^*_0a$, linear in time; store $\phi_i,\psi_i$ and $\sum_i\bar\phi_i$
  \item The element advances $\phi_{3D}$ (Eq.~34 with the front term \fterm{}), $c$ and $a$, consumed by
        $g\,\phi_{3D}(\sum_i\bar\phi_i)$
\end{enumerate}
\end{frame}

\begin{frame}[label=bk1]{Backup 1: stationarity conditions of the point model}
\small
$\dot\Psi+\Delta+C\to\operatorname{stat}$ with respect to $\dot\phi_i$, $\dot\psi_i$, $\dot\phi_0$ and $\gamma$, where
\[
\Psi=-\tfrac12c^*\bar{\bm\phi}^{\mathrm T}\mathbf A\,\bar{\bm\phi}
+\tfrac12\alpha^*\bm\psi^{\mathrm T}\mathbf B\,\bm\psi,\qquad
\Delta=\tfrac12\dot{\bar{\bm\phi}}^{\mathrm T}\bm\eta\,\dot{\bar{\bm\phi}}
+\tfrac12\dot{\bm\phi}^{\mathrm T}\bm\eta\,\dot{\bm\phi},\qquad
C=\gamma\Bigl(\sum_{l=0}^n\phi_l-1\Bigr)
\]
gives, with $\bar\phi_i=\phi_i\psi_i$ and $\dot{\bar\phi}_i=\dot\phi_i\psi_i+\phi_i\dot\psi_i$,
\[
\begin{aligned}
0&=-c^*\psi_i(\mathbf A\bar{\bm\phi})_i+\eta_i\bigl(\dot\phi_i\psi_i^2+\bar\phi_i\dot\psi_i+\dot\phi_i\bigr)+\gamma,\\
0&=-c^*\phi_i(\mathbf A\bar{\bm\phi})_i+\alpha^*b_i\psi_i+\eta_i\bigl(\dot\psi_i\phi_i^2+\bar\phi_i\dot\phi_i\bigr),\\
0&=\gamma+\dot\phi_0,\qquad 0=\textstyle\sum_{l=0}^n\phi_l-1 .
\end{aligned}
\]
\begin{itemize}
  \item $2n+2$ equations; implicit Euler and Newton, $\Delta t=10^{-4}$; a logarithmic
        barrier keeps $\phi_i,\psi_i,\phi_0$ in $(0,1)$
  \item The second term of $\Delta$ is needed: the first alone gives a singular system
  \item Only the symmetric part of $\mathbf A$ enters $\Psi$ $\Rightarrow$ $\mathbf A=\mathbf A^{\mathrm T}$
\end{itemize}
\end{frame}

\begin{frame}[label=bk17]{Backup 17: what is solved where, in one step}
\small
\cbox{p2}{1.45cm}{\textbf{\color{p2}1. Mechanics: nodes, global matrix (ANSYS, Newton)}\\[-4pt]
\[
\mathbf K_T\,\Delta\mathbf u=\mathbf f_{\mathrm{ext}}-\mathbf f_{\mathrm{int}}(\mathbf u),\qquad
\mathbf f_{\mathrm{int}}=\textstyle\sum_e\int_{\Omega_e}\mathbf B^{\mathrm T}\bm\sigma\,dV,\qquad
\bm\sigma\ \text{from}\ \mathbf F_e=\mathbf F\mathbf F_g^{-1},\ \mathbf F_g=\alpha\mathbf I
\]}\\[3pt]
\cbox{p1}{2.35cm}{\textbf{\color{p1}2. Fields: Gauss points $p$, explicit, no solve}\\[-6pt]
\[
\begin{aligned}
\phi_p^{n+1}&=\phi_p^n+\Delta t\Big[\beta\textstyle\sum_q L_{pq}\phi_q+k_\alpha\alpha_p
 -r\frac{c_p}{k+c_p}\,\mathbf n_c\cdot\sum_q\mathbf G_{pq}\phi_q\Big]\\[-2pt]
c_p^{n+1}&=c_p^n+\Delta t\Big[d\textstyle\sum_q L_{pq}c_q-g\,\phi_p\big(\sum_i\bar\phi_i\big)_p\,c_p\Big]\qquad(a\ \text{alike})
\end{aligned}
\]
\vspace{-4pt}\scriptsize $L_{pq}$, $\mathbf G_{pq}$: Laplacian and gradient weights from the 30 neighbours (NEM)}\\[3pt]
\cbox{c3}{1.3cm}{\textbf{\color{c3}3. Point model: each Gauss point alone, $12\times12$ (Newton, backward Euler)}\\[-4pt]
\[
\mathbf R(\mathbf x_p^{n+1};\,c^*,\alpha^*)=\mathbf 0,\qquad
\mathbf J\,\Delta\mathbf x=-\mathbf R,\quad \mathbf J=\partial\mathbf R/\partial\mathbf x,\qquad
\mathbf x=(\phi_1..\phi_5,\phi_0,\psi_1..\psi_5,\gamma)
\]}
\takeaway{Only $\mathbf K_T$ is a global matrix; the fields need products, the point model a small matrix per point, like a plastic strain.}
\end{frame}
