\begin{frame}{Appendix B: background, growth field and mechanics}
\small
\textbf{Kinematics.} Growth is a volume change: $\mathbf F=\mathbf F_e\mathbf F_g$ with
isotropic $\mathbf F_g=(1+\alpha)\mathbf I$. Stress comes only from $\mathbf F_e$ (neo-Hookean).
A body that can grow freely stays stress free; stress appears where growth is constrained.\\
\textbf{Material} (the example input, solver units MPa): $E_{\rm bio}=1000$, $E_{\rm void}=1$,
$\nu=0.3$, blended linearly with the local biofilm fraction; no viscosity.
Stresses scale linearly with $E$.
\vspace{4pt}

\textbf{Field equations} (Klempt et al.\ 2024, from an extended Hamilton principle):
\[
\dot\varphi=\underbrace{\beta\,\Delta\varphi}_{\text{diffusion}}
+\underbrace{k_a\,\alpha_K}_{\text{local source}}
-\underbrace{r\,\frac{c}{K+c}\,\nabla\varphi\cdot\frac{\nabla c}{|\nabla c|}}_{\text{front growth towards nutrient}},
\qquad
\dot\alpha_K=k_a\,\varphi
\]
\begin{itemize}
  \item $\varphi$ and $c$ are nodal fields of the element; $\alpha$ is an internal
        variable at the Gauss points
  \item The front term moves the biofilm boundary up the nutrient gradient; it vanishes
        where $\varphi$ or $c$ is uniform
  \item The nutrient diffuses much faster than the biofilm grows (quasi-static $c$)
\end{itemize}
\end{frame}

\begin{frame}{Appendix C: background, point model and one coupling step}
\small
\textbf{Point model} (Klempt et al.\ 2026), per species $i$, no space dependence:
\[
\begin{aligned}
(\eta_{\varphi,i}+\eta_i\psi_i^2)\,\dot\varphi_i+\eta_i\varphi_i\psi_i\,\dot\psi_i
&= c^*\psi_i\textstyle\sum_j A_{ij}\varphi_j\psi_j-\gamma-(\text{barrier})\\
\eta_i\,(\varphi_i\psi_i\,\dot\varphi_i+\varphi_i^2\,\dot\psi_i)
&= c^*\varphi_i\textstyle\sum_j A_{ij}\varphi_j\psi_j-\alpha^* b_i\,\psi_i-(\text{barrier})
\end{aligned}
\]
with $\sum_i\varphi_i+\varphi_0=1$; the barrier terms keep $\varphi_i,\psi_i$ in $(0,1)$.
Implicit time stepping, $\Delta t=10^{-4}$.
\vspace{4pt}

\textbf{One time step of the coupled ANSYS run}
\begin{enumerate}
  \item The element updates its field $\varphi$ (Eq.\ 34)
  \item At each Gauss point: $\varphi_{3D}\to\alpha$ (Eq.\ 36) $\to\mathbf F_g\to$ stress
  \item Two species: the point model is scaled to $\min(\varphi_{3D},\varphi_{\rm cap})$,
        advanced by $s\,\Delta t$, scaled again; $\chi_i,\psi_i$ are stored as state variables
\end{enumerate}
\end{frame}
