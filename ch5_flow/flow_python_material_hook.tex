% Zoom-in on the dashed "coupling layer" box of flow_impl_architecture: what one
% Fortran<->Python material evaluation actually sends and returns.
% Grounded in the implemented wire protocol (ansys_usermat/coupling/protocol.py,
% material_server.py), not a generic viscoplastic-UMAT template — see the two
% callouts for where this law's specifics diverge from that template.
% Usage: \resizebox{\linewidth}{!}{\input{flow_python_material_hook.tex}}
\begin{tikzpicture}[
  every node/.style = {font=\small},
  N/.style   = {draw, rounded corners=3pt, align=left, thick,
                text width=9.2cm, inner sep=6pt},
  Nio/.style = {N, fill=teal!12},
  Nint/.style= {N, fill=orange!12},
  ar/.style  = {-Stealth, thick},
  lbl/.style = {font=\scriptsize\itshape, fill=white, inner sep=1.5pt, align=center},
  call/.style= {draw, rounded corners=3pt, align=left, thick, inner sep=4pt,
                fill=gray!8, font=\scriptsize, text width=5.0cm},
  lead/.style= {-Stealth, thick, gray!55},
]

% ── Fortran call ──────────────────────────────────────────────────────
\node[Nio] (in) {%
  \textbf{Fortran call} — one Gauss point, one increment\\
  \texttt{UMAT} (Abaqus) / \texttt{USERMAT} (ANSYS), \texttt{kUsePy=1}\\[4pt]
  \begin{tabular}{@{}p{2.85cm}p{5.85cm}@{}}
  \texttt{F} & total deformation gradient $\mathbf F$ (3$\times$3), $d\mathbf x=\mathbf F\,d\mathbf X$\\
  \texttt{Fv} & converged viscous grad.\ $\mathbf F_v^n$ from step $t_n$ (3$\times$3)\\
  \texttt{alpha} & growth driver $\alpha$ — \emph{prescribed}, from the JAXFEM field\\
  \texttt{dt} & increment $\Delta t=t_{n+1}-t_n$\\[2pt]
  \texttt{C10, C01,}\newline\texttt{D1, eta, mtype} & Mooney--Rivlin / viscosity constants (ANSYS \texttt{prop} array)
  \end{tabular}};

\node[below=0.5cm of in] (req) {\textit{ISO\_C\_BINDING / local socket — newline-JSON request}};
\draw[ar] (in) -- (req);

% ── Python local integration ────────────────────────────────────────
\node[Nint, below=0.5cm of req] (py) {%
  \textbf{Python material hook — local integration at this point}\\
  (\texttt{material\_server.py})\\[5pt]
  1.\ elastic trial: $\mathbf F_e^{\mathrm{trial}}=\mathbf F_{\mathrm{mech}}(\mathbf F_v^n)^{-1}$,
     $\mathbf F_{\mathrm{mech}}=\mathbf F\,\mathbf F_g^{-1}$, $\mathbf F_g=(1{+}\alpha)\mathbf I$\\[2pt]
  2.\ viscous update (biofilm Standard-Solid flow rule), flow increment at $\mathbf F_v^n$\\[2pt]
  3.\ recompute $\mathbf F_e$, $J_e=\det\mathbf F_e$ at $\mathbf F_v^{n+1}$\\[2pt]
  4.\ Cauchy stress $\bm\sigma(\mathbf F_e)$ — Neo-Hookean/Mooney--Rivlin deviator
     $+\,D_1$ pressure\\[2pt]
  5.\ consistent tangent \texttt{dsdePl} $=\partial\bm\sigma/\partial\bm\varepsilon$
     by symmetric $\mathbf F$-perturbation\\[-1pt]
  \hspace*{1.1em}\small(6 evaluations, same scheme the verified UMAT uses)};

\node[below=0.5cm of py] (resp) {\textit{newline-JSON response}};
\draw[ar] (py) -- (resp);

% ── Return to Fortran ────────────────────────────────────────────────
\node[Nio, below=0.5cm of resp] (out) {%
  \textbf{Return to Fortran}\\[5pt]
  \begin{tabular}{@{}ll@{}}
  \texttt{stress} & Cauchy $\bm\sigma$, 6$\times$1 Abaqus Voigt (11,22,33,12,13,23)\\
   & $\to$ element $f^{\mathrm{int}}=\int B^{\!\top}\bm\sigma\,dV$\\[4pt]
  \texttt{Fv\_new} & $\mathbf F_v^{n+1}$ (3$\times$3) $\to$ stored in \texttt{ustatev}/\texttt{STATEV}
     for $t_{n+2}$\\[4pt]
  \texttt{dsdePl} & 6$\times$6 tangent $\to$ element stiffness
     $K_e=\int B^{\!\top}\mathbb C\,B\,dV$
  \end{tabular}};

% ── global Newton–Raphson loop-back, routed clear of all boxes ───────
\coordinate (dropL) at ($(in.west)+(-2.2,0)$);
\draw[ar] (out.west) -| (dropL) |- (in.west);
\node[lbl, anchor=east, text width=2.7cm, align=right]
     at ($(dropL)+(-0.15,0)$)
     {global Newton--Raphson: assemble $K_e,f^{\mathrm{int}}$, check residual,
      quadratic convergence};

% ── callouts: where this law diverges from a generic template ────────
\node[call, right=1.9cm of out, anchor=west, yshift=-0.3cm] (c1) {%
  \textbf{no \texttt{alpha\_new} in the response}\\
  Unlike a generic viscoplastic hook, $\alpha$ here is not a locally-integrated
  hardening variable — it is \emph{prescribed} per point from the JAXFEM
  $\alpha$-field (the offline hand-off in \texttt{flow\_impl\_architecture}).
  Only $\mathbf F_v$ is local internal state, so \texttt{protocol.py}
  correctly returns \texttt{stress, Fv\_new, detFe, dsdePl} — no
  $\alpha_{\mathrm{new}}$.};
\draw[lead] (out.east) -- (c1.west);

\node[call, right=1.9cm of py, anchor=west, yshift=0.2cm] (c2) {%
  \textbf{\texttt{dsdePl} convention, fixed}\\
  Solvers differ on what tangent they expect (Jaumann-rate elastoplastic,
  Truesdell/$\partial P/\partial F$, \dots). This hook always returns the
  $\mathbf F$-perturbation tangent the verified UMAT core uses — confirmed
  correct \emph{because} \texttt{SOLID185/NLGEOM,ON} converges with it
  (\texttt{ansys\_usermat/apdl/}).};
\draw[lead] (py.east) -- (c2.west);

\end{tikzpicture}
