\begin{tikzpicture}[
  every node/.style = {font=\small},
  chk/.style  = {draw, rounded corners=3pt, align=left, thick, fill=teal!10,
                 text width=6.4cm, inner sep=6pt},
  why/.style  = {draw, rounded corners=3pt, align=left, thick, fill=gray!7,
                 text width=6.9cm, inner sep=6pt, font=\footnotesize},
  gap/.style  = {draw, rounded corners=3pt, align=left, very thick,
                 draw=red!65!black, fill=red!7, text width=13.9cm, inner sep=7pt},
  ar/.style   = {-Stealth, thick, gray!55},
]

\node[gap] (top) {%
  \textbf{The error none of the four sees}\\[3pt]
  The isochoric split applies $J^{-2/3}$ to the subtracted trace but not to the
  tensor beside it. The coded and correct forms agree exactly at $J=1$ and
  nowhere else --- and growth is what moves $J_e$ away from $1$. The resulting
  discrepancy is \textbf{purely spherical}: a pressure error, with the
  deviatoric and von Mises stress untouched to machine precision.};

\node[chk, below left=1.0cm and -3.2cm of top.south] (c1) {%
  \textbf{1. Two implementations, 0 ULP}\\
  \scriptsize 8017 states, found by adversarial search};
\node[why, right=0.5cm of c1] (w1) {%
  Both ports carry \emph{the same expression}. Agreement between them is
  agreement, not correctness --- the distinction this section exists to make.};

\node[chk, below=0.35cm of c1] (c2) {%
  \textbf{2. Free-growth closed form}\\
  \scriptsize traction-free expansion, stress $\equiv 0$};
\node[why, right=0.5cm of c2] (w2) {%
  Traction-free growth gives $\mathbf F_e = \mathbf I$ exactly, so $J_e = 1$ ---
  precisely the one case where both forms agree.};

\node[chk, below=0.35cm of c2] (c3) {%
  \textbf{3. Constrained-growth references}\\
  \scriptsize the values shipped with the decks};
\node[why, right=0.5cm of c3] (w3) {%
  Generated by \emph{calling} the implementation, so they record its behaviour
  rather than test it. Now relabelled a regression snapshot.};

\node[chk, below=0.35cm of c3] (c4) {%
  \textbf{4. Dual-UMAT growth comparison}\\
  \scriptsize different elastic potentials, agree to 0.1\%};
\node[why, right=0.5cm of c4] (w4) {%
  Compares \emph{von Mises}, which cannot see a spherical term. Blind by
  construction, not by accident --- and still valid for what it was built for.};

\foreach \i in {1,2,3,4} { \draw[ar] (c\i) -- (w\i); }
\draw[ar] (top.south) -- ++(0,-0.45);

\node[below=0.45cm of c4.south west, anchor=north west, align=left,
      font=\footnotesize\itshape, text width=13.9cm, gray!45!black] {%
  The fifth check closed the gap: a closed form derived independently of the code, which reproduces
  code, which reproduces the
  implementation's output as \emph{closed form $+$ the predicted spherical
  term} to $10^{-12}$ --- pinning both the code and the explanation of the
  discrepancy at once.};
\end{tikzpicture}
