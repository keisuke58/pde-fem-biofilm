\begin{frame}{最初の30秒：この修論で何をしたか}
\small\Q{この修論の新しいところは何か}
\A{点モデルの組成を、FEM の成長場の各 Gauss 点に入れたこと。
場が量を、点モデルが組成を決める。}
\begin{enumerate}\footnotesize
  \item \textbf{点モデルの較正}（ch3）：5菌種・4条件の in-vitro データから、相互作用 15 個を
        TMCMC で推定。較正に使っていない pH を $R^2=0.78$ で予測
  \item \textbf{成長則を ANSYS に}（ch4）：$\mathbf F=\mathbf F_e\mathbf F_g$、
        $\mathbf F_g=\alpha\mathbf I$（Klempt 2024）をユーザー材料に。閉じた形の解と比べて検証。
        IKM の要素の場に $\dot\alpha=k_\alpha\phi$ でつないだ
  \item \textbf{各 Gauss 点に組成}（ch4）：2菌種の点モデル（Klempt 2026）を場と並べて解く。
        ANSYS からの呼び出しは Python とビット単位で一致
\end{enumerate}
\vspace{4pt}
\small\Q{一言でいうと、何ができるようになったのか}
\A{「何でできているか」の較正された記述と、「どう育ち、どんな応力を持つか」の
FEM の記述が、1菌種と2菌種でつながった。}
\end{frame}

\begin{frame}[shrink=5]{Junker 先生（1）：変分原理}
\small\Q{点モデルの式を、原理から導いてください}
\A{$\dot\Psi+\Delta+C\to\operatorname{stat}$（$\dot{\bm\phi},\dot{\bm\psi},\gamma$ について）。
ホワイトボードで書けるようにしておく。}
\begin{itemize}\footnotesize
  \item $\Psi=-\tfrac12c^*\bar{\bm\phi}^{\!T}\mathbf A\bar{\bm\phi}$、
        $\bar\phi_i=\phi_i\psi_i$；$\Delta$ は速度の二次形式（$\eta_i$）；
        $C=\gamma(\sum_{l=0}^n\phi_l-1)$
  \item $\dot\phi_i$ で微分 → 体積分率の式、$\dot\psi_i$ → 生存率の式、$\gamma$ → 体積の拘束
  \item 未知数 $2n+2$ 個、陰的 Euler と Newton 法
\end{itemize}
\vspace{3pt}
\small\Q{最小散逸ポテンシャルの原理と、拡張 Hamilton 原理の関係は}
\A{点モデル（変位がない、慣性もない）では、前者が後者の\textbf{局所形}。}
\begin{itemize}\footnotesize
  \item Junker, Makowski \& Hackl 2014、Hackl 1997。修論 ch2 に書いた
  \item 場のモデル（ch4）は変位があるので Hamilton 原理の形（Klempt 2024 Eq.~19）
\end{itemize}
\vspace{3pt}
\small\Q{$\mathbf A$ が対称なのは、仮定なのか結果なのか}
\A{結果。エネルギーが $\bar{\bm\phi}$ の二次形式なので、反対称部分は式に現れない。}
\begin{itemize}\footnotesize
  \item その代わり、\textbf{向きのある相互作用}（付着、毒素）は表せない。限界として ch3 に書いた
\end{itemize}
\end{frame}

\begin{frame}{Junker 先生（2）：成長則とつなぎ方}
\small\Q{$\mathbf F_g=\alpha\mathbf I$ で、等方でよいのか}
\A{Klempt 2024 に従った。異方性を決めるデータがない。}
\vspace{2pt}
\small\Q{Klempt 2024 には $\alpha$ の則が2つある。どちらを使ったか}
\A{\textbf{Eq.~36}（$\dot\alpha=k_\alpha\phi$）。Table~1 の形とは違う結果になることも示した（付録）。}
\vspace{2pt}
\small\Q{点モデルと場の時間はどうつないだのか}
\A{\textbf{仮定}。点モデルは1サブステップに $s\,\Delta t$ 進む（$s=0.15$）。
論文には書かれていないので、感度を表にした（$s=0.05$〜$1$）。}
\begin{itemize}\footnotesize
  \item もう一つの仮定は上限 $\phi_{\rm cap}=0.9$：点モデルには $\min(\phi_{3D},\phi_{\rm cap})$ を渡す。
        これも感度つき（0.85〜0.95）
\end{itemize}
\vspace{2pt}
\small\Q{組成は応力に効くのか}
\A{\textbf{効かない}（一方向のつなぎ）。Klempt 2024 が成長の式の力学の項を落としているのに従った。}
\begin{itemize}\footnotesize
  \item 組成から剛性へ、応力から成長へ、は今後の課題（Keio WP2：恒常圧の成長則の試作はある）
\end{itemize}
\end{frame}

\begin{frame}[shrink=3]{Junker 先生（3）：NEM と時間刻み}
\small\Q{場の更新は陽解法。時間刻みの条件は}
\A{Rudolf et al.\ 2025 の $\Delta t\le h^2/(2\beta)$ は\textbf{1次元の条件}（$\lambda\le1/2$）。
積分点の格子で NEM の stencil を Bloch 解析すると、限界はメッシュに依存：\textbf{8$^3$ で 0.06、16$^3$ で 0.14〜0.15、24$^3$ で 0.16〜0.17}（付録 D）。実行はこれをはさむ。}
\begin{itemize}\footnotesize
  \item 理由：重み $\exp[-\frac12(4d/\beta^*)^2]$ の幅 $\beta^*=0.2$\,mm が固定なので、粗いメッシュほど近い点だけで当てはめ、係数が大きくなる
  \item 重みなしの極限は 0.18（格子が不等間隔なので 7点差分の 1/6 と同じにならない）
  \item 「最後まで走った」は安定の判定にならない（0.160 は止まらずに振動した）
\end{itemize}
\vspace{3pt}
\small\Q{要素の中で見つけた問題は}
\A{表面の点の $\phi$ が更新されず、どの面も $\phi=0$ の境界になっていた。}
\begin{itemize}\footnotesize
  \item 直して回し直すと、主な結果（シードの応力、$\alpha$、組成）の変化は $0.5\,\%$ 以内
  \item $\beta$ が大きいときだけシードの圧力が大きく動く（$\beta=0.1$ で約 $30\,\%$）。表は直した値
  \item 修論では、誰の誤りかではなく、何が起きていたかと直し方だけを書いた
\end{itemize}
\vspace{3pt}
\small\Q{$\beta=0.02$\,mm$^2/T^*$ はどこから}
\A{Klempt 2024 Table~2 の $\beta=2$ を 2\,mm の立方体に換算した。\textbf{仮定}として書き、
$\beta=0.005$〜$0.1$ の感度を表にした。}
\end{frame}

\begin{frame}{Wangenheim 先生（1）：工学的な意味}
\small\Q{応力が $10^{-4}$\,Pa のオーダー。意味があるのか}
\A{剛性が小さい（$E=10$\,Pa、Klempt 2024 Table~2）ので、応力も小さい。
見るべきは\textbf{大きさより分布}：シードは圧縮、周りは引張。}
\begin{itemize}\footnotesize
  \item 周りの引張の輪は、Klempt et al.\ が報告したものと合う
  \item 剛性が $E\phi^2$ に比例するので、応力は剛性に比例して変わる
\end{itemize}
\vspace{3pt}
\small\Q{実験で確かめたのか}
\A{\textcolor{copen}{\textbf{していない。}}検証（verification）はしたが、
妥当性確認（validation）はしていない。}
\begin{itemize}\footnotesize
  \item 成長する口腔バイオフィルムの応力の測定値が手に入らなかった
  \item 確かめたのは：閉じた形の解、勾配のない領域の厳密解（$\alpha$ で $3\times10^{-11}$）、
        論文のモデルとの比較
  \item 組成の側は in-vitro データで較正し、較正に使っていない pH で確かめた
\end{itemize}
\vspace{3pt}
\small\Q{なぜ ANSYS なのか}
\A{IKM の要素（Neighbored Element Method）が ANSYS 用だから。}
\end{frame}

\begin{frame}{Wangenheim 先生（2）：数値の信頼性}
\small\Q{メッシュに対して収束しているか}
\A{例題のメッシュは、シードの平均応力を\textbf{約 1.7 倍}大きく見積もる。限界として書いた。}
\begin{itemize}\footnotesize
  \item 角のピークはメッシュによる。平均の値で議論している
  \item 16$^3$ から 24$^3$ では、シードの応力が $3$〜$8\,\%$、$\alpha$ が平均 $0.6\,\%$ で変わる
        （修論後の WP3 の結果）。32$^3$ はメモリ不足で回らない
\end{itemize}
\vspace{3pt}
\small\Q{Newton 法は収束しているか}
\A{各サブステップ 1〜4 回の反復で収束。自動時間刻みは一度も刻みを縮めなかった。}
\vspace{3pt}
\small\Q{時間刻みを半分にするとどうなるか}
\A{シードの平均応力が約 $1\,\%$ 変わる（$\Delta t=0.025$ と $0.0125$）。}
\vspace{3pt}
\small\Q{成長則の実装は正しいか}
\A{1要素で、自由と拘束の両方を閉じた形の値と比べた。拘束した場合
$\sigma=-1.0193\times10^{-4}$ が一致。}
\end{frame}

\begin{frame}[shrink=6]{Soleimani 先生・Geisler 先生：較正（ch3）}
\small\Q{なぜ TMCMC なのか}
\A{事後分布が多峰で、サンプルごとに並列にできるから。}
\begin{itemize}\footnotesize
  \item JAX で GPU に載せて、CPU の約 150 倍（500 粒子の比較で $147\times$）
\end{itemize}
\vspace{2pt}
\small\Q{$c^*=25$ に固定したのはなぜ}
\A{$c^*$ は $c^*A_{ij}$ の積でしか式に入らないので、それだけでは決まらない。
$A_{ij}$ が $\mathcal O(1)$ になるように固定した。}
\vspace{2pt}
\small\Q{Phase 1 と Phase 2 の違いは}
\A{Phase 1 は $\psi$ を固定、Phase 2 は $\psi$ も動かす。常在の静置では MAP がよく合い（$r=0.87$）、
dysbiotic では \textit{Pg} に関わる係数がずれる（Fig.~3.9）。}
\vspace{2pt}
\small\Q{条件の差は本当に生態の差か}
\A{一部は\textbf{菌株の差}。常在と dysbiotic で \textit{P.~gingivalis} の株が違う（DSM 20709 と W83）。}
\vspace{2pt}
\small\Q{予測は何か}
\A{\textit{F.~nucleatum} を除くと、dysbiotic の終盤の \textit{Pg} の増加が抑えられる
（$\hat a_{45}$ が dysbiotic の流れ条件でだけ大きい）。共培養で確かめられる。}
\end{frame}

\begin{frame}{答えが「分からない」「していない」もの}
\small ここは\textbf{言い切る練習}をしておく。取り繕うと必ず深掘りされる。
\vspace{4pt}
\begin{itemize}\footnotesize
  \item \textbf{応力の validation}：\textcolor{copen}{していない}。測定値がない
  \item \textbf{前線の項}：ANSYS の要素では\textcolor{copen}{働いていない}。
        バイオフィルムは拡散だけで広がる。組成は局所の栄養で変わるが、栄養へ向かう成長では変わらない
  \item \textbf{Klempt 2024 の再現}：test case 4.1 は、書いた\textbf{ずれ}つきで再現（消費 $g\phi c$）。
        印刷された式そのままでは再現できない（付録）
  \item \textbf{向きのある相互作用}：対称な $\mathbf A$ では\textcolor{copen}{表せない}
  \item \textbf{3菌種以上、組成から剛性へ、応力から成長へ}：今後（慶應）
\end{itemize}
\vspace{6pt}
\small\Q{どこまで信じてよいのか}
\A{数値としては、検証した範囲で信じてよい。生物の予測としては、pH という独立な量で
一度確かめただけ。応力の絶対値は、剛性の値に比例してそのまま動く。}
\end{frame}
