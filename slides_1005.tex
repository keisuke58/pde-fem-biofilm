\begin{frame}{The coupling: what the element does and what is added}
\small
\textbf{The element (Oliver's, unchanged)} solves the biofilm field of
Klempt et al.\ 2024, Eq.\ 34, at its points (front term: see ``found along the way''):
\[
\dot\phi-\beta\,\Delta\phi-k_\alpha\,\alpha
+|\nabla\phi|\,r\,\frac{c}{K+c}\,\mathbf n_\phi\!\cdot\mathbf n_c=0,
\qquad \dot\alpha=k_\alpha\,\phi\quad\text{(Eq.\ 36)}
\]
\textbf{Added in this work}, at every Gauss point of the material routine:
\begin{itemize}
  \item take $\phi$ from the field and integrate Eq.\ 36:
        $\alpha_{n+1}=\alpha_n+k_\alpha\,\phi\,\Delta t$
  \item growth $\mathbf F_g=\alpha\mathbf I$ $\to$ stress (verified law, previous slide)
\end{itemize}
\vspace{2pt}
Parameters: Klempt et al.\ 2024 Table 2, $k_\alpha=10^{-3}$ per unit time $T^*$,
in the field \emph{and} in Eq.\ 36; $E=10$ Pa, $\nu=0.49$, stiffness $\propto\phi^2$.
ANSYS 2022 R2, one-species runs.
\vspace{4pt}

\begin{frame}{Four things found along the way}
\small
\begin{enumerate}
  \item \textbf{The element's growth variable is the mean of the two species fields.}
        With one species, the unused field stays at its start value, so growth is halved.
        $\Rightarrow$ question to Oliver: intended?
  \item \textbf{Start values must satisfy $\phi_1+\phi_2\le 1$}
        (the void fraction is $\phi_0=1-\sum\phi_i$). The example input starts
        both species at 1 in the seed; for one species I start the second at 0.
  \item \textbf{The first result set is not in equilibrium:} the element returns zero
        stress on its first calls while it sets up its operators. This set is excluded.
  \item \textbf{The element's front-growth term is inactive:} a copy slip in reading
        its weights sets the weight of species 1 to zero. It also uses
        $|\Delta\phi|$ where Eq.\ 34 has $|\nabla\phi|$. With both corrected in a
        test copy the seed spreads, but $\phi$ exceeds 1. Checks 1--3 do not
        depend on this term. $\Rightarrow$ to discuss with Oliver.
\end{enumerate}
\vspace{6pt}
\colorbox{c3!12}{\parbox{0.95\linewidth}{\small
Earlier bring-up runs used larger growth rates (0.1, 0.5) and the example input's
stiffness (1000 MPa); all results on these slides use the published values.}}
\end{frame}

\begin{frame}{Two modelling assumptions, stated explicitly}
\small
\begin{enumerate}
  \item \textbf{Time link $s$.} Klempt et al.\ 2024 run to $T^*=1$; Klempt et al.\ 2026 run
        500--1500 steps of $10^{-4}$ in its own unit. No paper links the two.
        The point model advances $s\,\Delta t$ per step; here $s=0.15$.
        \begin{itemize}
          \item case 6: the winner emerges only for $s\gtrsim0.15$
          \item case 3: $s$ moves $\phi_1/(\phi_1+\phi_2)$ by 0.03--0.06
        \end{itemize}
  \item \textbf{Packed regions, $\phi_{\rm cap}=0.9$.} Where the field is fully
        packed ($\phi_{3D}\to1$) the point model sees at most 0.9; otherwise the
        result depends on the coupling time step. Sensitivity: 0.664--0.668 (case 3).
\end{enumerate}
\vspace{6pt}
All other parameters are the published values.
\end{frame}

  \item \textbf{Fix:} one shared interface mesh. The bonded layer bulges

  \item \textbf{Consequence:} growth layer $6.3\times$ the substrate von Mises

\begin{frame}{Appendix B: background, growth field and mechanics}
\small
\textbf{Kinematics.} Growth is a volume change: $\mathbf F=\mathbf F_e\mathbf F_g$ with
isotropic $\mathbf F_g=\alpha\mathbf I$. Stress comes only from $\mathbf F_e$ (neo-Hookean).
A body that can grow freely stays stress free; stress appears where growth is constrained.\\
\textbf{Material} (Klempt et al.\ 2024, Eq.\ 20 and Table 2): $E=10$ Pa, $\nu=0.49$,
stiffness weighted with $\phi^2$ (void: floor $10^{-3}E$); no viscosity.
Stresses scale linearly with $E$ (the example input's 1000 MPa gives $10^8\times$ larger values).
\vspace{4pt}

\textbf{Field equations} (Klempt et al.\ 2024, from an extended Hamilton principle):
\[
\dot\phi=\underbrace{\beta\,\Delta\phi}_{\text{diffusion}}
+\underbrace{k_\alpha\,\alpha}_{\text{local source}}
-\underbrace{r\,\frac{c}{K+c}\,\nabla\phi\cdot\frac{\nabla c}{|\nabla c|}}_{\text{front growth towards nutrient}},
\qquad
\dot\alpha=k_\alpha\,\phi
\]
\begin{itemize}
  \item $\phi$ and $c$ are nodal fields of the element; $\alpha$ is an internal
        variable at the Gauss points
  \item The front term moves the biofilm boundary up the nutrient gradient; it vanishes
        where $\phi$ or $c$ is uniform
  \item The nutrient diffuses much faster than the biofilm grows (quasi-static $c$)
\end{itemize}
\end{frame}

\begin{frame}{Appendix C: background, point model and one coupling step}
\small
\textbf{Point model} (Klempt et al.\ 2026), per species $i$, no space dependence:
\[
\begin{aligned}
(\eta_{\phi,i}+\eta_i\psi_i^2)\,\dot\phi_i+\eta_i\phi_i\psi_i\,\dot\psi_i
&= c^*\psi_i\textstyle\sum_j A_{ij}\phi_j\psi_j-\gamma-(\text{barrier})\\
\eta_i\,(\phi_i\psi_i\,\dot\phi_i+\phi_i^2\,\dot\psi_i)
&= c^*\phi_i\textstyle\sum_j A_{ij}\phi_j\psi_j-\alpha^* b_i\,\psi_i-(\text{barrier})
\end{aligned}
\]
with $\sum_i\phi_i+\phi_0=1$; the barrier terms keep $\phi_i,\psi_i$ in $(0,1)$.
Implicit time stepping, $\Delta t=10^{-4}$.
\vspace{4pt}

\textbf{One time step of the coupled ANSYS run}
\begin{enumerate}
  \item The element updates its field $\phi$ (Eq.\ 34)
  \item At each Gauss point: $\phi_{3D}\to\alpha$ (Eq.\ 36) $\to\mathbf F_g\to$ stress
  \item Two species: the point model is scaled to $\min(\phi_{3D},\phi_{\rm cap})$,
        advanced by $s\,\Delta t$, scaled again; $\phi_i,\psi_i$ are stored as state variables
\end{enumerate}
\end{frame}
