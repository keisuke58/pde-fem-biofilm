  \item Eq.\ 34 with front term, Eq.\ 36 unchanged

\begin{frame}{Equations}
\begin{align*}
c^*(t)&=c^*_0\Bigl[\tfrac{t_{n+1}-t}{\Delta t}c_n+\tfrac{t-t_n}{\Delta t}c_{n+1}\Bigr],\quad \alpha^*\ \text{likewise}\\
\dot c&=d\,\nabla^2c-g\,\phi_{3D}\,\textstyle\sum_i\bar\phi_i\,c,\qquad \bar\phi_i=\phi_i\psi_i\\
\dot a&=d_a\nabla^2a-g_a\,\phi_{3D}\,\textstyle\sum_i\bar\phi_i\,a\\
\dot\phi_{3D}&=\beta\nabla^2\phi_{3D}+k_\alpha\alpha-\|\nabla\phi_{3D}\|\,\tfrac{r\,c}{k+c}\,\mathbf n_{\nabla\phi_{3D}}\cdot\mathbf n_{\nabla c},
\qquad \dot\alpha=k_\alpha\phi_{3D}
\end{align*}
\small $g\phi$ in Eq.\ 35 of the paper is a misprint; $g\phi_{3D}c$ is used (Monod form $f(c)=c/(K_c+c)\to c/K_c$ for $K_c\gg c$). $\phi_{3D}$: field, $\phi_i$: point model.
\end{frame}

\begin{itemize}\setlength{\itemsep}{4pt}
  \item No interpolation from nodes: the fields live at the integration points
  \item The antibiotic in the second nutrient field takes the same path
  \item One increment behind: the fields are solved after the material routine
  \item A short first load step solves the fields before the point model starts
\end{itemize}
\end{frame}

\begin{frame}{First ANSYS run: what is checked}
\centering
\includegraphics[width=\linewidth,height=0.45\textheight,keepaspectratio]{assets/fig_living_trace_check.png}\\[2pt]
\raggedright\small
\begin{itemize}\setlength{\itemsep}{2pt}
  \item (a) $c_{n+1}$ of one increment is $c_n$ of the next: the pair is passed on without a jump
  \item (b) point model in ANSYS = Python with the same inputs: $2\times10^{-12}$ or better in every step
        (four runs, with and without antibiotic)
  \item The first step needs up to 38 Newton iterations from the start state; with a tolerance-controlled
        Newton (up to 50) both codes agree there too
\end{itemize}
\end{frame}
